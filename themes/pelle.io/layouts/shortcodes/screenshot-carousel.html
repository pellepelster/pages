{{- range $index, $item := $images }} {{- end }}
{{- range $index, $item := $images }} {{- $image := $.Page.Resources.Get (printf "%s/%s" $dir $item.file) -}} {{- if not $image -}} {{- errorf "screenshot-carousel: %s/%s not found in %s" $dir $item.file $.Page.File.Path -}} {{- end -}} {{- $slide := $image.Resize "1200x webp" -}} {{- $full := $image.Resize (printf "%dx webp" $image.Width) -}}
